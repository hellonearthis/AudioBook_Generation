function getRelativeLuminance(r: number, g: number, b: number): number {

}

function getContrastRatio(lum1: number, lum2: number): number {
    const L1 = Math.max(lum1, lum2);
    const L2 = Math.min(lum1, lum2);
    return (L1 + 0.05) / (L2 + 0.05);
}
